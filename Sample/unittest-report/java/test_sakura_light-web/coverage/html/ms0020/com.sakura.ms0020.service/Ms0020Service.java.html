    /** Resolve the read key from the screen field, falling back to the current record's key. */

        }
        if (key == null || key.trim().isEmpty()) {
            try { key = file.extractKeyFromCurrentRecord(); } catch (Exception _e3) {}
        }
        return key != null ? key.trim() : "";
