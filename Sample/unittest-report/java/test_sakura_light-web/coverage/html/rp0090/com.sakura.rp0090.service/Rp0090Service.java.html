    /**

     * key extracted from the current record buffer if that field is blank.

    private void readByKeyWithFallback(RawDatasetBase file, String keyFieldName) {
        String rkVal = "";
        if (rkVal == null || rkVal.trim().isEmpty()) {
          try { rkVal = ws.getString(keyFieldName); } catch (Exception _e) {}
        }
        if (rkVal == null || rkVal.trim().isEmpty()) {
          try { rkVal = file.extractKeyFromCurrentRecord(); } catch (Exception _e3) {}
        }
        file.readByKey(rkVal != null ? rkVal.trim() : "");
        ws.trySetString("FSTS", file.getFileStatus());
    }
