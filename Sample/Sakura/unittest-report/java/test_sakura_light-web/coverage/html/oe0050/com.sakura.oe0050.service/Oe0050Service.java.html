    /**

          try { rkVal = ws.getString(keyField); } catch (Exception _e) {}
        }
        if (rkVal == null || rkVal.trim().isEmpty()) {

        }
        file.readByKey(rkVal != null ? rkVal.trim() : "");
        ws.trySetString("FSTS", file.getFileStatus());
    }
